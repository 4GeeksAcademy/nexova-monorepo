==================== DESPUÉS: sudo limitado de deploy ====================

$ cat /etc/sudoers.d/90-nexova-deploy
# Gestionado por infra/server/bootstrap.sh — no editar a mano.
# 'deploy' NO está en el grupo docker (equivaldría a root): solo puede
# ejecutar estos comandos exactos. Sin comodines para impedir argumentos extra
# (p. ej. 'docker compose -f /otro.yml' o 'run --privileged').
Cmnd_Alias NEXOVA_COMPOSE = /usr/bin/docker compose --project-directory /srv/nexova/app pull, \
                            /usr/bin/docker compose --project-directory /srv/nexova/app up -d, \
                            /usr/bin/docker compose --project-directory /srv/nexova/app up -d --remove-orphans, \
                            /usr/bin/docker compose --project-directory /srv/nexova/app down, \
                            /usr/bin/docker compose --project-directory /srv/nexova/app ps, \
                            /usr/bin/docker compose --project-directory /srv/nexova/app restart
Cmnd_Alias NEXOVA_SYSTEMCTL = /usr/bin/systemctl status docker, \
                              /usr/bin/systemctl restart docker, \
                              /usr/bin/systemctl status ssh, \
                              /usr/bin/systemctl reload ssh
Defaults:deploy !visiblepw, use_pty, logfile=/var/log/sudo-nexova.log
deploy ALL=(root) NOPASSWD: NEXOVA_COMPOSE, NEXOVA_SYSTEMCTL
